গুগল শিটস সম্পূর্ণ ডাটাবেস সিঙ্ক (Orders + Expenses + Stock + Menu)

আপনার গুগল শিটের Extensions > Apps Script-এ নিচের কোডটি পেস্ট করে Web App (Anyone) হিসেবে Deploy করুন এবং প্রাপ্ত URL-টি নিচে সেভ করুন।

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  function readTab(name) {
    var sh = ss.getSheetByName(name);
    if (!sh) return [];
    var rows = sh.getDataRange().getValues();
    var arr = [];
    for (var i = 1; i < rows.length; i++) {
      if (rows[i][0]) { try { arr.push(JSON.parse(rows[i][0])); } catch(err){} }
    }
    return arr;
  }
  return ContentService.createTextOutput(JSON.stringify({
    orders: readTab("Orders"),
    expenses: readTab("Expenses"),
    stock: readTab("Stock"),
    menu: readTab("Menu")
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var payload = JSON.parse(e.postData.contents);
  function writeTab(name, data, headers, mapper) {
    var sh = ss.getSheetByName(name) || ss.insertSheet(name);
    sh.clear();
    sh.appendRow(headers);
    (data || []).forEach(function(item) { sh.appendRow(mapper(item)); });
  }
  writeTab("Orders", payload.orders, ["JSON", "OrderID", "Date", "Customer", "Total", "Status"], function(o) {
    return [JSON.stringify(o), o.id, o.date, o.customerName, o.total, o.status];
  });
  writeTab("Expenses", payload.expenses, ["JSON", "Date", "Category", "Description", "Amount"], function(ex) {
    return [JSON.stringify(ex), ex.date, ex.category, ex.desc, ex.amount];
  });
  writeTab("Stock", payload.stock, ["JSON", "ItemName", "Quantity", "Unit"], function(st) {
    return [JSON.stringify(st), st.name, st.qty, st.unit];
  });
  writeTab("Menu", payload.menu, ["JSON", "ItemName", "Category", "Price"], function(mn) {
    return [JSON.stringify(mn), mn.name, mn.category, mn.price];
  });
  return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
}
